function deepQueryAll(selector, root = document) {
  let results = [...root.querySelectorAll(selector)];
  root.querySelectorAll('*').forEach(el => {
    if (el.shadowRoot) results = results.concat(deepQueryAll(selector, el.shadowRoot));
  });
  return results;
}

async getFieldsFromSection(section, dataFieldId) {
    const rootElement = await section.getRootElement();

    // real column tag, not the wildcard content/getContent() - avoids the over/under-match
    const columnElements = await rootElement.findElements(
        By.css("[slot='columns'] flexipage-column2")
    );

    let allFields = [];
    for (const columnElement of columnElements) {
        // filter by data-field-id directly in the selector, not after casting every field -
        // only elements matching this specific id get queried/wrapped at all
        const fieldElements = await columnElement.findElements(
            By.css(`flexipage-field[data-field-id='${dataFieldId}']`)
        );
        const fields = await Promise.all(
            fieldElements.map(el => createInstance(Field, field.driver, el))
        );
        allFields = allFields.concat(fields);
    }

    return allFields;
}

async getsFieldsAcrossSections(sections, fieldMappingV2) {
    const allFields = new Map();
    for (const section of sections) {
        const sectionFields = await this.getFieldsFromSection(section, fieldMappingV2);
        for (const [fieldName, field] of sectionFields) {
            allFields.set(fieldName, field);
        }
    }
    return allFields;
}

async getFieldsFromSection(section, fieldMappingV2) {
    const rootElement = await section.getRootElement();
    const columnElements = await rootElement.findElements(
        By.css("flexipage-column2[slot='columns']")
    );

    const allFields = new Map();
    for (const columnElement of columnElements) {
        for (const [fieldName, fieldId] of Object.entries(fieldMappingV2)) {
            // findElements (plural) returns [] instead of throwing when this column
            // doesn't contain this particular field - most won't, per column
            const fieldElements = await columnElement.findElements(
                By.css(`flexipage-field[data-field-id='${fieldId}']`)
            );
            if (fieldElements.length === 0) continue;

            const field = await createInstance(Field, section.driver, fieldElements[0]);
            allFields.set(fieldName, field);
        }
    }
    return allFields;
}
