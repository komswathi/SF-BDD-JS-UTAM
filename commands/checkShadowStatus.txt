function checkShadowStatus(tagList, root = document.body) {
  const targets = new Set(tagList.map(t => t.toLowerCase()));
  const rows = [];

  function walk(node) {
    if (node.nodeType !== 1) return;
    const tag = node.tagName.toLowerCase();

    if (targets.has(tag)) {
      rows.push({
        tag,
        found: true,
        shadow: !!node.shadowRoot
      });
    }

    if (node.shadowRoot) {
      for (const child of node.shadowRoot.children) walk(child);
    }
    for (const child of node.children) walk(child);
  }

  walk(root);

  const foundTags = new Set(rows.map(r => r.tag));
  for (const t of targets) {
    if (!foundTags.has(t)) {
      rows.push({ tag: t, found: false, shadow: null });
    }
  }

  console.table(rows);
  return rows;
}

checkShadowStatus(['lightning-input', 'lightning-combobox']);
